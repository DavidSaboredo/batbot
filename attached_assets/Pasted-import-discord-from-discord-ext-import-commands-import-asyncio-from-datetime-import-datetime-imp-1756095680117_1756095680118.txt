@bot.event
async def on_voice_state_update(member, before, after):
    """Evento que se ejecuta cuando hay cambios en el estado de voz de un usuario"""
    
    # Verificar si el usuario se unió a un canal válido
    if after.channel and after.channel.id in VALID_VOICE_CHANNELS:
        # Usuario se unió a un canal válido
        if member.id not in user_join_times:
            user_join_times[member.id] = datetime.now()
            print(f'{member.display_name} se unió al canal {after.channel.name} a las {user_join_times[member.id].strftime("%H:%M:%S")}')
    
    # Verificar si el usuario salió de un canal válido
    if before.channel and before.channel.id in VALID_VOICE_CHANNELS:
        # Usuario salió de un canal válido
        if member.id in user_join_times:
            # Calcular tiempo de conexión
            join_time = user_join_times[member.id]
            leave_time = datetime.now()
            connection_duration = (leave_time - join_time).total_seconds()
            
            print(f'{member.display_name} salió del canal {before.channel.name} después de {connection_duration:.0f} segundos')
            
            # Verificar qué rol le corresponde según el tiempo de conexión
            await check_and_assign_role(member, connection_duration)
            
            # Remover del diccionario
            del user_join_times[member.id]
    
    # Caso especial: usuario se movió de un canal válido a otro canal válido
    if (before.channel and before.channel.id in VALID_VOICE_CHANNELS and 
        after.channel and after.channel.id in VALID_VOICE_CHANNELS and 
        before.channel.id != after.channel.id):
        # No hacer nada, mantener el tiempo de entrada original
        print(f'{member.display_name} se movió de {before.channel.name} a {after.channel.name} (manteniendo tiempo de entrada)')

async def check_and_assign_role(member, connection_duration):
    """Verifica qué rol le corresponde al usuario según su tiempo de conexión"""
    # Determinar el rol más alto que puede obtener
    eligible_role = None
    for role_config in reversed(ROLE_SYSTEM):  # Empezar por el rol más alto
        if connection_duration >= role_config["time_seconds"]:
            eligible_role = role_config
            break
    
    if eligible_role:
        await assign_progressive_role(member, eligible_role, connection_duration)
    else:
        min_time = ROLE_SYSTEM[0]["time_seconds"]
        print(f'{member.display_name} no cumplió el tiempo mínimo ({connection_duration:.0f}/{min_time} segundos)')

async def assign_progressive_role(member, role_config, connection_duration):
    """Asigna el rol progresivo correspondiente al usuario"""
    try:
        role_name = role_config["name"]
        
        # Buscar el rol en el servidor
        role = discord.utils.get(member.guild.roles, name=role_name)
        
        if not role:
            # Si el rol no existe, crearlo
            print(f'El rol "{role_name}" no existe. Creándolo...')
            role = await member.guild.create_role(
                name=role_name,
                color=role_config["color"],
                reason="Rol creado automáticamente por el sistema de roles progresivos"
            )
            print(f'Rol "{role_name}" creado exitosamente')
        
        # Verificar si el usuario ya tiene este rol o uno superior
        current_role_level = get_user_role_level(member)
        new_role_level = get_role_level(role_name)
        
        if current_role_level >= new_role_level:
            print(f'{member.display_name} ya tiene el rol "{role_name}" o uno superior')
            return
        
        # Remover roles anteriores del sistema progresivo
        await remove_previous_roles(member)
        
        # Asignar el nuevo rol
        await member.add_roles(role, reason=f'Tiempo acumulado: {connection_duration:.0f} segundos')
        print(f'✅ Rol "{role_name}" asignado a {member.display_name} (tiempo: {connection_duration:.0f}s / {connection_duration/60:.1f}min)')
        print(f'   → {role_config["description"]}')
        
    except discord.Forbidden:
        print(f'❌ Error: No tengo permisos para asignar roles a {member.display_name}')
    except discord.HTTPException as e:
        print(f'❌ Error HTTP al asignar rol a {member.display_name}: {e}')
    except Exception as e:
        print(f'❌ Error inesperado al asignar rol a {member.display_name}: {e}')

def get_role_level(role_name):
    """Obtiene el nivel del rol en el sistema progresivo"""
    for i, role_config in enumerate(ROLE_SYSTEM):
        if role_config["name"] == role_name:
            return i
    return -1

def get_user_role_level(member):
    """Obtiene el nivel más alto del usuario en el sistema progresivo"""
    max_level = -1
    for role in member.roles:
        level = get_role_level(role.name)
        if level > max_level:
            max_level = level
    return max_level

async def remove_previous_roles(member):
    """Remueve roles anteriores del sistema progresivo"""
    roles_to_remove = []
    for role in member.roles:
        if any(role.name == role_config["name"] for role_config in ROLE_SYSTEM):
            roles_to_remove.append(role)
    
    if roles_to_remove:
        await member.remove_roles(*roles_to_remove, reason="Actualización de rol progresivo")
        print(f'Roles anteriores removidos de {member.display_name}: {", ".join([r.name for r in roles_to_remove])}')

@bot.command(name='voice_stats')
@commands.has_permissions(administrator=True)
async def voice_stats(ctx):
    """Comando para ver estadísticas de usuarios conectados (solo administradores)"""
    if not user_join_times:
        await ctx.send('No hay usuarios conectados en canales monitoreados.')
        return
    
    stats_message = "**Usuarios actualmente en canales monitoreados:**\n"
    current_time = datetime.now()
    
    for user_id, join_time in user_join_times.items():
        user = bot.get_user(user_id)
        if user:
            duration = (current_time - join_time).total_seconds()
            stats_message += f"• {user.display_name}: {duration:.0f} segundos ({duration/60:.1f} minutos)\n"
    
    await ctx.send(stats_message)

@bot.event
async def on_command_error(ctx, error):
    if isinstance(error, commands.MissingPermissions):
        await ctx.send('❌ No tienes permisos para usar este comando.')
    elif isinstance(error, commands.CommandNotFound):
        pass  # Ignorar comandos no encontrados
    else:
        print(f'Error en comando: {error}')
